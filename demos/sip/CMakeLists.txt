# ---------------------------------------------------------------------------
# doppler-sip - the SIP-mode demo.
#
# Uses Pulse purely as a media engine; SIP signalling is handled by PJSIP.
# See README.md for the flow.
#
# Self-contained, like pexninja/: the parent CMakeLists.txt just calls
# add_subdirectory(demos/sip) when BUILD_DOPPLER_SIP=ON.
# ---------------------------------------------------------------------------

# ---- System dependencies --------------------------------------------------
# OpenGL / GLFW / Threads come from the parent's find_package() calls
# (they're cached). PkgConfig is used to find PJSIP.
find_package(PkgConfig REQUIRED)

# PJSIP is not in the Ubuntu archive — see scripts/install-pjsip.sh in the
# repo root, which fetches+builds+installs it under /usr/local. After that
# pkg-config picks it up as `libpjproject`.
pkg_check_modules(PJPROJECT REQUIRED IMPORTED_TARGET libpjproject)
message(STATUS "Found PJSIP via pkg-config: ${PJPROJECT_VERSION}")

# PJSIP's pjlib uses libuuid for GUID generation when uuid.h is present at
# build time, but its libpjproject.pc doesn't propagate `-luuid` to consumers.
# Probe libuuid via pkg-config (preferred — apt: `uuid-dev`) and fall back to
# searching for the bare library if .pc isn't available.
pkg_check_modules(UUID IMPORTED_TARGET uuid)
if(NOT UUID_FOUND)
    find_library(UUID_LIBRARY NAMES uuid)
    if(NOT UUID_LIBRARY)
        message(FATAL_ERROR
            "libuuid not found - PJSIP's pjlib links against uuid_generate(). "
            "Install it with:  sudo apt-get install uuid-dev")
    endif()
    message(STATUS "Found libuuid: ${UUID_LIBRARY}")
endif()

# ---- Dear ImGui (reuses the parent's FetchContent declaration) ------------
# The parent CMakeLists.txt already calls FetchContent_MakeAvailable(imgui),
# so imgui_SOURCE_DIR is in scope here. We compile our own little imgui
# static lib so we don't tangle our backend choice with anyone else's.
if(NOT DEFINED imgui_SOURCE_DIR)
    message(FATAL_ERROR
        "imgui_SOURCE_DIR not set - demos/sip/ must be added from the parent "
        "CMakeLists.txt after FetchContent_MakeAvailable(imgui).")
endif()

add_library(doppler_sip_imgui STATIC
    ${imgui_SOURCE_DIR}/imgui.cpp
    ${imgui_SOURCE_DIR}/imgui_draw.cpp
    ${imgui_SOURCE_DIR}/imgui_tables.cpp
    ${imgui_SOURCE_DIR}/imgui_widgets.cpp
    ${imgui_SOURCE_DIR}/backends/imgui_impl_glfw.cpp
    ${imgui_SOURCE_DIR}/backends/imgui_impl_opengl3.cpp)
target_include_directories(doppler_sip_imgui PUBLIC
    ${imgui_SOURCE_DIR}
    ${imgui_SOURCE_DIR}/backends)
target_link_libraries(doppler_sip_imgui PUBLIC glfw OpenGL::GL)

# ---- The doppler-sip executable -------------------------------------------
add_executable(doppler-sip
    src/main.cpp
    src/sip_ua.cpp
    src/app_transport.cpp)

target_link_libraries(doppler-sip PRIVATE
    doppler_sip_imgui
    pexip::pulse
    PkgConfig::PJPROJECT
    $<IF:$<BOOL:${UUID_FOUND}>,PkgConfig::UUID,${UUID_LIBRARY}>
    Threads::Threads)

# Same RPATH treatment as the parent's `doppler` target so libpexpulse.so
# resolves without the user setting LD_LIBRARY_PATH for the direct dep.
set_target_properties(doppler-sip PROPERTIES
    BUILD_RPATH   "${PEXIP_PREFIX}/lib"
    INSTALL_RPATH "${PEXIP_PREFIX}/lib")

# Convenience launcher (mirrors run-doppler.sh) - sets LD_LIBRARY_PATH so
# Pulse's transitive private siblings (libpexlgpl.so, libimf.so, ...) are
# resolvable at runtime.
file(GENERATE
    OUTPUT  "${CMAKE_BINARY_DIR}/run-doppler-sip.sh"
    CONTENT "#!/bin/sh
# Auto-generated by CMake - sets LD_LIBRARY_PATH so libpexpulse.so's
# private siblings are resolvable, then execs the demo binary.
export LD_LIBRARY_PATH=\"${PEXIP_PREFIX}/lib:\${LD_LIBRARY_PATH}\"
exec \"$<TARGET_FILE:doppler-sip>\" \"$@\"
"
    FILE_PERMISSIONS OWNER_READ OWNER_WRITE OWNER_EXECUTE
                     GROUP_READ GROUP_EXECUTE
                     WORLD_READ WORLD_EXECUTE)
