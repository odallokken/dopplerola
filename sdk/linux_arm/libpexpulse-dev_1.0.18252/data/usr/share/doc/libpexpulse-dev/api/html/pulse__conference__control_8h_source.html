57struct _PulseConferenceControlDialRequest
58{
59 /* Refer to https://docs.pexip.com/api_client/api_rest.htm?Highlight=rest#dial for full descriptions of the struct
60 * members. */
61 /* Mandatory struct members */
62 PulseConferenceRole role; /* The level of privileges the participant has in the conference. */
63 const char * destination; /* The target address to call. */
64 PulseConferenceProtocol protocol; /* The protocol to use to place the outgoing call. */
65
66 /* Optional struct members */
67 const char * presentation_url; /* This additional parameter can be specified for RTMP calls to send the presentation
68 stream to a separate RTMP destination. */
69 bool streaming; /* Identifies the dialed participant as a streaming or recording device. [True: streaming/recording
70 participant | False: not a streaming/recording participant]. */
71 const char *
72 dtmf_sequence; /* An optional DTMF sequence to transmit after the call to the dialed participant starts. */
73 const char * source_display_name; /* Optional field that specifies what the calling display name should be. */
74 const char * source; /* Optional field that specifies the source URI (must be a valid URI for the conference). */
75 PulseConferenceCallType call_type; /* Optional field that limits the media content of the call. */
76 PulseConferenceKeepAliveType keep_alive_type; /* Determines whether the conference continues when all other non-ADP
77 participants have disconnected */
78 const char * remote_display_name; /* An optional friendly name for this participant. This may be used instead of the
79 participant's alias in participant lists and as a text overlay in some layout
80 configurations. */
81 const char * text; /* Optional text to use instead of remote_display_name as the participant name overlay text. */
82};
